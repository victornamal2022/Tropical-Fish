// --- Módulo 1 (Parte B): Pedidos por Cliente SIN despachar --- function dibujarPedidosPendientes() { // Filtramos donde la cantidad despachada (D) sea menor que la solicitada (C) // o donde el estado (B) sea 'Pendiente por Despachar' var query = new google.visualization.Query( `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?sheet=Detalle_pedidos&tq=select A, E, C, D where B = 'Pendiente por Despachar'` ); query.send(function(response) { if (response.isError()) { console.error('Error Módulo Pedidos Pendientes: ' + response.getMessage()); return; } var table = new google.visualization.Table(document.getElementById('mod_1_pedidos_pendientes')); // Asegúrate de crear este div en el HTML table.draw(response.getDataTable(), {width: '100%', height: '100%', page: 'enable', pageSize: 10}); }); } // --- Módulo 6: Cumplimiento de Cliente (Despachado vs Solicitado) --- function dibujarModuloCumplimiento() { // Agrupamos por Cliente (A), sumando lo despachado (D) y lo solicitado (C) var query = new google.visualization.Query( `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?sheet=Detalle_pedidos&tq=select A, sum(D), sum(C) group by A label sum(D) 'Total Despachado', sum(C) 'Total Solicitado'` ); query.send(function(response) { if (response.isError()) { console.error('Error Módulo Cumplimiento: ' + response.getMessage()); return; } var data = response.getDataTable(); // Añadimos una columna calculada dinámicamente para el Porcentaje (%) data.addColumn('number', '% Cumplimiento'); // Iteramos sobre las filas devueltas para calcular el porcentaje exacto for (var i = 0; i < data.getNumberOfRows(); i++) { var despachado = data.getValue(i, 1); // sum(D) var solicitado = data.getValue(i, 2); // sum(C) // Evitar división por cero var porcentaje = solicitado > 0 ? (despachado / solicitado) : 0; // Asignamos el valor numérico y su formato visual en porcentaje data.setValue(i, 3, porcentaje); data.setFormattedValue(i, 3, (porcentaje * 100).toFixed(1) + '%'); } var table = new google.visualization.Table(document.getElementById('mod_6_cumplimiento')); // Usamos un DataView para mostrar solo Cliente, y el % de Cumplimiento si queremos resumir // O lo dejamos completo pasándole 'data' directamente table.draw(data, {width: '100%', height: '100%', allowHtml: true, showRowNumber: true}); }); }
